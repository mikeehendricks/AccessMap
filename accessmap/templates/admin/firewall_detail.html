Forwarding examples

FortiGate (syslog over HTTPS JSON forwarder)

config log syslogd setting
    set server "{{ ingest_url }}"
    set format json
    set custom-field-name1 "Authorization"
    set custom-field-value1 "Bearer {{ token_example }}"
end
config log syslogd filter
    set advanced-filter "field['action'] eq 'allow' or field['action'] eq 'deny'"
end
